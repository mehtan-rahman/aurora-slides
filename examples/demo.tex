\section{Mathematics}

\begin{frame}{Equations are written, stroke by stroke}
  Inline maths such as $e^{i\pi} + 1 = 0$ and $f\colon \R \to \R$ sits in the text.
  \pause
  \begin{equation}\label{eq:gauss}
    \int_{-\infty}^{\infty} e^{-x^2}\,dx = \sqrt{\pi}
  \end{equation}
  \pause
  \begin{align*}
    (a + b)^2 &= a^2 + 2ab + b^2 \\
    (a - b)^2 &= a^2 - 2ab + b^2
  \end{align*}
  \pause
  Equation~\eqref{eq:gauss} is numbered, and the reference follows it.
\end{frame}

\section{Animated scenes}

\begin{frame}{Plots driven by a parameter}
\begin{scene}[size={16,6}]
  \Axes[id=ax, at={-2,0}, w=9, h=4.4, x={0,10,2}, y={-1.2,1.2,0.6}]
  \ValueTracker[id=k, value=1]
  \pause
  \Plot[on=ax, color=blue]{sin(k*x)}
  \MathTex[at={5.2,1.4}, size=44, with]{y = \sin(kx)}
  \Text[at={4.6,0.3}, anchor=left, size=30, color=muted, with]{k =}
  \DecimalNumber[at={5.6,0.3}, value={k}, digits=2, size=44, with]
  \pause
  \Animate[k=3, run=2.5]
  \pause
  \Plot[on=ax, color=orange]{0.6*sin(k*x/2)*exp(-0.15*x)}
  \Animate[k=1.5, run=2, with]
\end{scene}
  \note{Anything that mentions a ValueTracker redraws while it animates.}
\end{frame}

\begin{frame}{Reaction mechanisms}
\begin{scene}[size={16,6.2}]
  \Molecule[id=nu, at={-4.4,0.4}, lp=auto, bond=1.6]{[OH-:5]}
  \Molecule[id=r, at={0.2,0}, lp={3}, bond=1.6, with]{[CH3:1][C:2](=[O:3])[Cl:4]}
  \pause
  \CurlyArrow[from=lp:5, to=2]
  \CurlyArrow[from=2=3, to=3]
  \pause
  \Transform[id={r,nu}, lp={3:3}]{[CH3:1][C:2]([O-:3])([Cl:4])[OH:5]}
  \Text[id=lbl, at={2.1,1.8}, anchor=left, color=muted, size=26]{tetrahedral intermediate}
  \pause
  \CurlyArrow[from=lp:3, to=2-3]
  \CurlyArrow[from=2-4, to=4]
  \pause
  \Transform[id=r, lp={3,4}]{[CH3:1][C:2](=[O:3])[OH:5].[Cl-:4]}
  \FadeOut[id=lbl, with]
  \Text[at={2.1,1.8}, anchor=left, color=muted, size=26]{substitution product}
\end{scene}
  \note{Molecules come from SMILES; curly arrows name atoms by their map numbers.}
  \note<3>{The tetrahedral centre is drawn in perspective: a wedge and a hash.}
\end{frame}
