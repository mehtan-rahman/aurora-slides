\section{Reaction coordinate}

\begin{frame}{Defining the committor}
  Suppose every partial trajectory carried one number. It would tell us:
  \begin{itemize}[<+->]
    \item which trajectories deserve more computation,
    \item where the boundary between not-yet-committed and committed lies,
    \item and \alert{how far} along the reaction we really are.
  \end{itemize}
  \pause
  \[ q(x) \;=\; \Pr\big[\,\tau_B < \tau_A \;\big|\; x(0) = x\,\big] \]
\end{frame}

\section{Central claim}

\begin{frame}{Equations are \emph{written}, not faded}
  The committor solves a boundary-value problem:
  \begin{align*}
    \mathcal{L}\,q(x) &= 0, && x \notin A \cup B,\\
    q\big|_A = 0, &\quad q\big|_B = 1 .
  \end{align*}
  \pause
  For a 1-D diffusion with potential $V(x)$ and temperature $\beta^{-1}$,
  \[ q(x) = \frac{\int_a^x e^{\beta V(y)}\,dy}{\int_a^b e^{\beta V(y)}\,dy}. \]
  \pause
  \begin{alertblock}{Takeaway}
    The committor is the \textbf{ideal reaction coordinate}: $q = \tfrac12$ is the transition-state surface.
  \end{alertblock}
\end{frame}
